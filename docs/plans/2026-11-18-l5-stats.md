## 1. Goal

Build the numbers lane as **pure functions** with unit tests: pass^k, Wilson intervals, Cohen's kappa for a judge, a flake quarantine contract, a trace-schema validator, and the release gates (golden-slice status, model-swap paired delta, prompt-regression band, cost and latency budgets, quota and safety statuses, resume cursor).

Every function takes plain data (`TrialRecord[]` and a few config values) and returns plain data. No function reads the clock, the environment, the network or the disk. The caller passes those in. That is what lets the same code run on L1 records today and on L2 to L4 records later.

```ts
// FILE: src/stats/status.ts
export type Mode = "stub" | "nightly";
export type TrialStatus = "pass" | "fail" | "partial" | "blocked by provider" | "expectedDefect" | "skipped";
export const TRIAL_STATUSES: readonly TrialStatus[] = ["pass", "fail", "partial", "blocked by provider", "expectedDefect", "skipped"];

/** P4_LLM_MODE. Unset or empty means stub. Anything else that is not stub or nightly throws (fail closed). */
export function parseMode(raw: string | undefined): Mode {
  if (raw === undefined || raw === "") return "stub";
  if (raw === "stub" || raw === "nightly") return raw;
  throw new Error(`P4_LLM_MODE must be stub or nightly, got: ${raw}`);
}

export function isRateLimited(text: string): boolean {
  return /\b429\b|RESOURCE_EXHAUSTED/.test(text);
}

export function isSafetyRefusal(text: string): boolean {
  return /content_filter|\bSAFETY\b/i.test(text);
}

export interface ProviderVerdict {
  status: "ok" | "partial" | "blocked by provider" | "fail";
  reason: string;
}

/** Same rules as tests/l1/gemma-classify.ts. A 429 is partial only when every retry was used. */
export function classifyProviderError(errorText: string, attempts: number, maxAttempts: number): ProviderVerdict {
  if (errorText === "") return { status: "ok", reason: "no provider error" };
  if (isRateLimited(errorText)) {
    return attempts >= maxAttempts
      ? { status: "partial", reason: `rate limited after ${attempts} attempt(s)` }
      : { status: "fail", reason: `rate limited after only ${attempts} of ${maxAttempts} attempts: retries not used up` };
  }
  if (isSafetyRefusal(errorText)) return { status: "blocked by provider", reason: "provider safety refusal" };
  return { status: "fail", reason: "provider error that is not a rate limit or a safety refusal" };
}
```

### T2. Wilson interval — `P4-STAT-004`

```ts
// FILE: src/stats/wilson.ts
import { ROUND_DP, Z_95 } from "./config.js";

export interface Interval {
  lower: number;
  upper: number;
}

/** Wilson score interval. Returns null when n is 0 (no data is not a rate). Throws RangeError on bad counts. */
export function wilsonInterval(successes: number, n: number, z: number = Z_95): Interval | null {
  if (!Number.isInteger(successes) || !Number.isInteger(n) || successes < 0 || n < 0 || successes > n) {
    throw new RangeError(`wilsonInterval needs integers with 0 <= successes <= n, got ${successes}/${n}`);
  }
  if (n === 0) return null;
  const p = successes / n;
  const z2 = z * z;
  const denom = 1 + z2 / n;
  const centre = (p + z2 / (2 * n)) / denom;
  const half = (z * Math.sqrt((p * (1 - p)) / n + z2 / (4 * n * n))) / denom;
  return { lower: Math.max(0, centre - half), upper: Math.min(1, centre + half) };
}

/** The committed rounding: half up at ROUND_DP places. Adding 0 turns -0 into 0 so JSON never shows "-0". */
export function round(x: number, dp: number = ROUND_DP): number {
  const f = 10 ** dp;
  return Math.round(x * f) / f + 0;
}

export function roundInterval(i: Interval): Interval {
  return { lower: round(i.lower), upper: round(i.upper) };
}

/** What artefacts publish: rate and interval, both rounded. Null when n is 0. */
export function publishedRate(successes: number, n: number): { rate: number; interval: Interval } | null {
  const w = wilsonInterval(successes, n);
  if (w === null) return null;
  return { rate: round(successes / n), interval: roundInterval(w) };
}
```

```ts
  caseId: string;
  trial: number;
  mode: Mode;
  status: TrialStatus;

export interface CaseScore {
  caseId: string;
  mode: Mode;
  k: number;
  coverage: Coverage;
  /** Trials with status pass or fail. */
  completed: number;
  passes: number;
  /** true only when coverage is complete and all k trials passed. null when not complete. 9/10 is false. */
  passK: boolean | null;
  /** passes / completed, rounded. null when nothing completed. */
  rate: number | null;
  /** Wilson interval over the completed trials (n = completed), rounded. null when nothing completed. */
  wilson: Interval | null;
  /** Trial indexes 0..k-1 with no record. Never silently dropped. */
  missingTrials: number[];
}

export function scoreCase(input: { caseId: string; mode: Mode; k: number }, trials: readonly TrialOutcome[]): CaseScore {
  const { caseId, mode, k } = input;
  if (!Number.isInteger(k) || k < 1) throw new ScoringError(`k must be a positive integer, got ${k}`);
  const seen = new Set<number>();
  for (const t of trials) {
    if (t.caseId !== caseId) throw new ScoringError(`trial for ${t.caseId} passed to ${caseId}`);
    if (t.mode !== mode) throw new MixedModeError(`${caseId}: trial ${t.trial} is ${t.mode}, expected ${mode}`);
    if (!Number.isInteger(t.trial) || t.trial < 0 || t.trial >= k) throw new ScoringError(`${caseId}: trial index ${t.trial} outside 0..${k - 1}`);
    if (seen.has(t.trial)) throw new ScoringError(`${caseId}: duplicate trial ${t.trial}`);
    seen.add(t.trial);
  }
  const missingTrials: number[] = [];
  for (let i = 0; i < k; i++) if (!seen.has(i)) missingTrials.push(i);

  const count = (s: TrialStatus): number => trials.filter((t) => t.status === s).length;
  const passes = count("pass");
  const completed = passes + count("fail");
  const partial = count("partial");
  const blocked = count("blocked by provider");
  const skipped = count("skipped");
  const defect = count("expectedDefect");

  let coverage: Coverage;
  if (trials.length > 0 && skipped === trials.length) coverage = "skipped";
  else if (trials.length > 0 && defect === trials.length) coverage = "expectedDefect";
  else if (skipped > 0 || defect > 0) throw new ScoringError(`${caseId}: skipped or expectedDefect mixed with measured trials`);
  else if (partial > 0 || missingTrials.length > 0) coverage = "partial"; // partial wins over blocked: it is resumable
  else if (blocked > 0) coverage = "blocked by provider";
  else coverage = "complete";

  const w = wilsonInterval(passes, completed);
  return {
    caseId,
    mode,
    k,
    coverage,
    completed,
    passes,
    passK: coverage === "complete" ? passes === k : null,
    rate: completed > 0 ? round(passes / completed) : null,
    wilson: w === null ? null : roundInterval(w),
    missingTrials,
  };
}
```

```ts
export function cohensKappa(pairs: readonly LabelPair[]): number | null {
  const n = pairs.length;
```

```ts
import { HEAVY_TURN_TOKENS, THROTTLE_WINDOW_MS, TPM_LIMIT, TPM_SAFETY } from "./config.js";

export interface ThrottleConfig {
  tpmLimit: number;
  safety: number;
  /** Tokens assumed for the next turn before its real usage is known. */
  estimateTokens: number;
  windowMs: number;
}

export const DEFAULT_THROTTLE: ThrottleConfig = {
  tpmLimit: TPM_LIMIT,
  safety: TPM_SAFETY,
  estimateTokens: HEAVY_TURN_TOKENS,
  windowMs: THROTTLE_WINDOW_MS,
};

export interface Throttle {
  /** Milliseconds to wait before the next heavy turn fits in the rolling window. 0 means go now. */
  waitMs(nowMs: number): number;
  /** Record real token usage after a turn. */
  record(tokens: number, nowMs: number): void;
}

/** Rolling-window token budget. The caller owns the clock and the sleeping, so this is a pure function of what it was told. */
export function createThrottle(cfg: ThrottleConfig = DEFAULT_THROTTLE): Throttle {
  const budget = Math.floor(cfg.tpmLimit * cfg.safety);
  if (cfg.estimateTokens > budget) throw new RangeError(`one estimated turn (${cfg.estimateTokens}) is larger than the budget (${budget})`);
  const used: Array<{ at: number; tokens: number }> = [];
  const prune = (now: number): void => {
    for (;;) {
      const first = used[0];
      if (first === undefined || first.at > now - cfg.windowMs) return;
      used.shift();
    }
  };
  return {
    waitMs(now) {
      prune(now);
      let total = used.reduce((sum, u) => sum + u.tokens, 0);
      if (total + cfg.estimateTokens <= budget) return 0;
      for (const u of used) {
        total -= u.tokens;
        if (total + cfg.estimateTokens <= budget) return Math.max(0, u.at + cfg.windowMs - now);
      }
      return cfg.windowMs;
    },
    record(tokens, now) {
      prune(now);
      used.push({ at: now, tokens });
    },
  };
}
```

### T9. Paired delta and model-swap gate — `P4-GATE-002`

Newcombe's hybrid-score interval for the difference of two paired proportions. With counts `a` = both true, `b` = only candidate true, `c` = only baseline true, `d` = neither, `n = a+b+c+d`, `p1 = (a+b)/n`, `p2 = (a+c)/n`, Wilson intervals `(l1,u1)` and `(l2,u2)` for `p1` and `p2`, delta `D = p1 - p2`:

- `A = (a+b)(c+d)(a+c)(b+d)` and `C = a*d - b*c`.
- phi = 0 when `A = 0`; else `(C - n/2) / sqrt(A)` when `C > n/2`; else 0 when `0 <= C <= n/2`; else `C / sqrt(A)`.
- lower = `D - sqrt((p1-l1)^2 + (u2-p2)^2 - 2*phi*(p1-l1)*(u2-p2))`
- upper = `D + sqrt((u1-p1)^2 + (p2-l2)^2 - 2*phi*(u1-p1)*(p2-l2))`

This is the formula that was run for finding F8.

```ts
// FILE: src/stats/paired.ts
import { round, wilsonInterval } from "./wilson.js";

export interface Pair {
  candidate: boolean;
  baseline: boolean;
}

export interface PairedDelta {
  n: number;
  both: number;
  onlyCandidate: number;
  onlyBaseline: number;
  neither: number;
  candidateRate: number;
  baselineRate: number;
  /** candidate minus baseline. */
  delta: number;
  lower: number;
  upper: number;
}

/** Paired difference of two proportions with a Newcombe hybrid-score 95% interval. null for no pairs. Values are rounded to the committed places. */
export function pairedDelta(pairs: readonly Pair[]): PairedDelta | null {
  const n = pairs.length;
  if (n === 0) return null;
  let a = 0;
  let b = 0;
  let c = 0;
  let d = 0;
  for (const p of pairs) {
    if (p.candidate && p.baseline) a++;
    else if (p.candidate) b++;
    else if (p.baseline) c++;
    else d++;
  }
  const p1 = (a + b) / n;
  const p2 = (a + c) / n;
  const w1 = wilsonInterval(a + b, n);
  const w2 = wilsonInterval(a + c, n);
  if (w1 === null || w2 === null) return null;
  const A = (a + b) * (c + d) * (a + c) * (b + d);
  const C = a * d - b * c;
  let phi = 0;
  if (A !== 0) {
    if (C > n / 2) phi = (C - n / 2) / Math.sqrt(A);
    else if (C < 0) phi = C / Math.sqrt(A);
  }
  const D = p1 - p2;
  const lo = D - Math.sqrt((p1 - w1.lower) ** 2 + (w2.upper - p2) ** 2 - 2 * phi * (p1 - w1.lower) * (w2.upper - p2));
  const hi = D + Math.sqrt((w1.upper - p1) ** 2 + (p2 - w2.lower) ** 2 - 2 * phi * (w1.upper - p1) * (p2 - w2.lower));
  return {
    n, both: a, onlyCandidate: b, onlyBaseline: c, neither: d,
    candidateRate: round(p1), baselineRate: round(p2),
    delta: round(D), lower: round(Math.max(-1, lo)), upper: round(Math.min(1, hi)),
  };
}
```

## 7. Running on L2, L3 and L4 records later

Nothing in `src/stats/` knows which lane produced a record. The contract for the other lanes:

1. A runner writes one `TrialRecord` per isolated trial (`schemaVersion: 1`, `caseId` from the ledger, `trial` 0 to k-1, `k`, `mode`). Use profile `state` when the case has a product-state oracle. That makes the before and after snapshots mandatory.
2. Records go to `evidence/traces/<caseId>/` and one line per record to `evidence/nightly/trials.jsonl`. `parseTrialsJsonl(text, "state")` is the loader.
3. A lane adds a measured case to the nightly slice with a one-line PR to `GOLDEN_SET` (and to `MEASURED_IDS` if a failed trial is a measured result and not a hard gate).
4. Hard security gates stay deterministic on stub: the case's own test asserts on product state and trajectory, then also validates its records with `validateTrial`. Measured assertions run on Gemma only.
5. A confirmed product bug is `expectedDefect` (the trial status exists for it). A deployment finding is an annotation.
6. L3 and L4 call `scoreCase`, `aggregateScores`, `buildAsrMatrix` and `budgetReport` on their own records. They do not copy the maths.

## 8. Throttle and one-lane-at-a-time (nightly)

- TPM (16,000, shared by the whole suite) is the real limit. D10 uses 12,000 usable per rolling 60 seconds: two heavy turns per minute.
- Only **one** slice runs per nightly job, and nightly jobs never run in parallel (`concurrency: nightly-gemma`, `cancel-in-progress: false`, no matrix). The slice name comes from the dispatch input. L3, L4 and L5 are sequenced by choosing the slice, never by running two Gemma jobs at once.
- On the first exhausted 429 (after 15 s and 45 s retries), the runner stops, marks the rest `partial` and writes the cursor. The job is green. The next dispatch resumes from the cursor.
- A safety-filter refusal is `blocked by provider` for that trial only. The slice continues.
- The free-tier numbers (30 RPM, 16K TPM, 14,400 RPD) come from third-party trackers and are unverified (master plan). The owner confirms them in AI Studio before the first nightly. Only `TPM_LIMIT` in `config.ts` changes if they differ.
